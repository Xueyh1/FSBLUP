cmake_minimum_required(VERSION 3.18)
project(fsblup VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(ARMADILLO_INCLUDE_DIR armadillo REQUIRED)
find_library(OPENBLAS_LIBRARY NAMES openblasp openblas REQUIRED)
add_library(fsblup_core STATIC src/io.cpp src/kernels.cpp src/solver.cpp src/search.cpp src/near_pd.cpp)
target_include_directories(fsblup_core PUBLIC src ${ARMADILLO_INCLUDE_DIR})
target_compile_definitions(fsblup_core PUBLIC ARMA_DONT_USE_WRAPPER ARMA_DONT_USE_OPENMP ARMA_DONT_USE_SUPERLU)
target_link_libraries(fsblup_core PUBLIC ${OPENBLAS_LIBRARY})
add_executable(fsblup src/main.cpp src/commands.cpp)
target_link_libraries(fsblup PRIVATE fsblup_core)
set_target_properties(fsblup PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
# Transitive bundled Fortran/OpenBLAS dependencies use the same relative path.
target_link_options(fsblup PRIVATE -Wl,--disable-new-dtags)
install(TARGETS fsblup DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_executable(core_test tests/core.cpp)
  target_link_libraries(core_test PRIVATE fsblup_core)
  add_test(NAME core COMMAND core_test ${CMAKE_CURRENT_SOURCE_DIR}/tests/data)
  add_test(NAME numerical_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check.py $<TARGET_FILE:fsblup>)
  add_test(NAME solver_shift COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/solver_shift.py $<TARGET_FILE:fsblup>)
  add_executable(nearpd_core_test tests/nearpd_core.cpp)
  target_link_libraries(nearpd_core_test PRIVATE fsblup_core)
  add_test(NAME nearpd_core COMMAND nearpd_core_test ${CMAKE_CURRENT_SOURCE_DIR}/tests/nearpd_cases.tsv)
  add_test(NAME nearpd_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/nearpd_cli.py $<TARGET_FILE:fsblup>)
  add_test(NAME nearpd_nested COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/nearpd_nested.py $<TARGET_FILE:fsblup>)
  add_test(NAME output_failure COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/output_failure.py $<TARGET_FILE:fsblup>)
  add_test(NAME solver_range COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/solver_range.py $<TARGET_FILE:fsblup>)
  set_tests_properties(core numerical_cli solver_shift nearpd_core nearpd_cli nearpd_nested output_failure solver_range PROPERTIES ENVIRONMENT "OPENBLAS_NUM_THREADS=1")
